// ============================================================ // ADMIN AUTHENTICATION // ============================================================ async function checkAdmin() { try { // -------------------------------------------------------- // 1. 현재 로그인한 사용자 확인 // -------------------------------------------------------- const { data: { user }, error: userError } = await supabaseClient.auth.getUser(); // 사용자 확인 오류 if (userError) { console.error( "Get user error:", userError ); return false; } // -------------------------------------------------------- // 2. 로그인하지 않은 경우 // -------------------------------------------------------- if (!user) { console.warn( "Admin check: User is not logged in." ); return false; } // -------------------------------------------------------- // 3. admin_users 테이블에서 관리자 확인 // -------------------------------------------------------- const { data, error } = await supabaseClient .from("admin_users") .select("id, email, role") .eq("user_id", user.id) .eq("role", "admin") .maybeSingle(); // Supabase DB 오류 if (error) { console.error( "Admin check error:", error ); return false; } // -------------------------------------------------------- // 4. 관리자 확인 // -------------------------------------------------------- if ( data && data.role === "admin" ) { console.log( "Admin verified:", data.email ); return true; } // -------------------------------------------------------- // 5. 관리자가 아닌 경우 // -------------------------------------------------------- console.warn( "Access denied: User is not an administrator." ); return false; } catch (error) { console.error( "Admin check failed:", error ); return false; } } // ============================================================ // PROTECT ADMIN PAGE // ============================================================ async function protectAdminPage() { try { const isAdmin = await checkAdmin(); if (!isAdmin) { alert( "Administrator access required." ); window.location.href = "/"; return false; } return true; } catch (error) { console.error( "Admin page protection failed:", error ); alert( "Unable to verify administrator access." ); window.location.href = "/"; return false; } } // ============================================================ // PAGE LOAD // ============================================================ document.addEventListener( "DOMContentLoaded", async () => { const allowed = await protectAdminPage(); if (!allowed) { return; } // ======================================================== // ADMIN ONLY CODE // ======================================================== console.log( "Admin page loaded." );
