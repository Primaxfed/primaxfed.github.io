-- ============================================================ -- PRIMAX GLOBAL -- ADMIN VENDOR MANAGEMENT RPC -- ============================================================ -- ============================================================ -- 1. ADMIN CHECK -- ============================================================ CREATE OR REPLACE FUNCTION public.is_admin() RETURNS boolean LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$ DECLARE result boolean; BEGIN SELECT EXISTS ( SELECT 1 FROM public.admin_users WHERE user_id = auth.uid() AND is_active = true ) INTO result; RETURN COALESCE(result, false); END; $$; -- ============================================================ -- 2. GET VENDOR LIST -- ============================================================ CREATE OR REPLACE FUNCTION public.admin_get_vendors() RETURNS TABLE ( vendor_id uuid, user_id uuid ) LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$ BEGIN -- ADMIN ONLY IF NOT public.is_admin() THEN RAISE EXCEPTION 'Not authorized'; END IF; RETURN QUERY SELECT DISTINCT vd.vendor_id, vd.user_id FROM public.vendor_documents vd WHERE vd.vendor_id IS NOT NULL ORDER BY vd.vendor_id; END; $$; -- ============================================================ -- 3. GET DOCUMENTS FOR SELECTED VENDOR -- ============================================================ CREATE OR REPLACE FUNCTION public.admin_get_vendor_documents( p_vendor_id uuid ) RETURNS SETOF public.vendor_documents LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$ BEGIN -- ADMIN ONLY IF NOT public.is_admin() THEN RAISE EXCEPTION 'Not authorized'; END IF; RETURN QUERY SELECT * FROM public.vendor_documents WHERE vendor_id = p_vendor_id OR user_id = p_vendor_id ORDER BY uploaded_at DESC; END; $$; -- ============================================================ -- 4. ADMIN INSERT DOCUMENT RECORD -- ============================================================ CREATE OR REPLACE FUNCTION public.admin_create_vendor_document( p_vendor_id uuid, p_user_id uuid, p_document_type text, p_file_name text, p_file_path text, p_file_size bigint, p_mime_type text ) RETURNS public.vendor_documents LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$ DECLARE new_document public.vendor_documents; BEGIN -- ADMIN ONLY IF NOT public.is_admin() THEN RAISE EXCEPTION 'Not authorized'; END IF; INSERT INTO public.vendor_documents ( vendor_id, user_id, document_type, file_name, file_path, file_size, mime_type, status ) VALUES ( p_vendor_id, p_user_id, p_document_type, p_file_name, p_file_path, p_file_size, p_mime_type, 'submitted' ) RETURNING * INTO new_document; RETURN new_document; END; $$; -- ============================================================ -- 5. PERMISSIONS -- ============================================================ GRANT EXECUTE ON FUNCTION public.is_admin() TO authenticated; GRANT EXECUTE ON FUNCTION public.admin_get_vendors() TO authenticated; GRANT EXECUTE ON FUNCTION public.admin_get_vendor_documents(uuid) TO authenticated; GRANT EXECUTE ON FUNCTION public.admin_create_vendor_document( uuid, uuid, text, text, text, bigint, text ) TO authenticated;
